-- [A3] 두 열: 열 개수 맞춰 합치기

-- [A4] 중복을 보존하는 1열 합치기

SELECT student_id FROM study_students WHERE department_id='D10' UNION ALL SELECT student_id FROM study_enrollments WHERE course_id='C01' ORDER BY student_id;

-- [B3] 원본에서 같은 ID가 이미 반복

SELECT student_id FROM study_enrollments WHERE student_id IN ('S101','S106') UNION SELECT student_id FROM study_students WHERE student_id='S101' ORDER BY student_id;

-- [B5] 두 열이면 행 전체로 중복 판정
