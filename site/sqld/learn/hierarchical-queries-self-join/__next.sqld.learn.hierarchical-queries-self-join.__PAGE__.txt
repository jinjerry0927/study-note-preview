-- [A1] 자신에게 상사를 연결하는 LEFT 셀프 조인
SELECT c.emp_id, c.emp_name, p.emp_name AS parent_name
FROM org_employees c LEFT JOIN org_employees p ON c.parent_id = p.emp_id ORDER BY c.emp_id;


-- [A2] INNER 셀프 조인은 루트를 제외
SELECT c.emp_id, c.emp_name, p.emp_id AS parent_id
FROM org_employees c JOIN org_employees p ON c.parent_id=p.emp_id ORDER BY c.emp_id;


-- [A3] 직속 자녀 0건도 세는 LEFT JOIN
SELECT p.emp_id, p.emp_name, COUNT(c.emp_id) AS direct_children
FROM org_employees p LEFT JOIN org_employees c ON c.parent_id=p.emp_id
GROUP BY p.emp_id,p.emp_name ORDER BY p.emp_id;


-- [A4] 같은 부모의 형제 노드 짝
SELECT a.emp_id AS left_id,b.emp_id AS right_id,a.parent_id
FROM org_employees a JOIN org_employees b ON a.parent_id=b.parent_id AND a.emp_id<b.emp_id
WHERE a.parent_id IS NOT NULL ORDER BY a.parent_id,left_id,right_id;

-- [B3] E110 아래 모든 후손
WITH RECURSIVE walk(emp_id, emp_name, parent_id, depth, root_id, root_name, id_path, name_path, sort_path) AS (
 SELECT emp_id, emp_name, parent_id, 1, emp_id, emp_name,
        '|'||emp_id||'|', '/'||emp_name,
        printf('%03d',sort_order)||'-'||emp_id
 FROM org_employees WHERE emp_id='E110'
 UNION ALL
 SELECT c.emp_id,c.emp_name,c.parent_id,w.depth+1,w.root_id,w.root_name,
        w.id_path||c.emp_id||'|',w.name_path||'/'||c.emp_name,
        w.sort_path||'/'||printf('%03d',c.sort_order)||'-'||c.emp_id
 FROM org_employees c JOIN walk w ON c.parent_id=w.emp_id
 WHERE w.depth<12 AND instr(w.id_path,'|'||c.emp_id||'|')=0 AND (1=1)
)
SELECT emp_id,emp_name,depth,name_path FROM walk ORDER BY sort_path;
