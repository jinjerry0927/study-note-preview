-- [B1] INNER: 양쪽 일치만

-- [B2] LEFT: 학생 모두 보존
SELECT s.student_id, d.department_id FROM study_students s LEFT OUTER JOIN study_departments d ON s.department_id = d.department_id ORDER BY s.student_id;

-- [B3] RIGHT: 학과 모두 보존
SELECT s.student_id, d.department_id FROM study_students s RIGHT OUTER JOIN study_departments d ON s.department_id = d.department_id ORDER BY d.department_id, s.student_id;

-- [B4] FULL: 학생·학과 양쪽 보존
SELECT s.student_id, d.department_id FROM study_students s FULL OUTER JOIN study_departments d ON s.department_id = d.department_id ORDER BY CASE WHEN s.student_id IS NULL THEN 1 ELSE 0 END, s.student_id, d.department_id;
