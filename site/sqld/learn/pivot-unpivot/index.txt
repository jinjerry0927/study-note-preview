DROP TABLE IF EXISTS sales_wide;
CREATE TABLE sales_long (
    sale_id TEXT PRIMARY KEY,
    region TEXT NOT NULL,
    channel TEXT NOT NULL,
    quarter TEXT NOT NULL,
    amount INTEGER
);

-- Wide data is a deliberately aggregated Q1~Q3 view, not a row-level original copy.
CREATE TABLE sales_wide AS
SELECT region,
       SUM(CASE WHEN quarter='Q1' THEN amount END) AS q1,
       SUM(CASE WHEN quarter='Q2' THEN amount END) AS q2,
       SUM(CASE WHEN quarter='Q3' THEN amount END) AS q3
FROM sales_long
GROUP BY region;

-- [A1] 원본 Long 11개 판매 기록
SELECT sale_id, region, channel, quarter, amount FROM sales_long ORDER BY sale_id;

-- [A2] 분기별 판매 원본 행 수
SELECT quarter, COUNT(*) AS fact_rows FROM sales_long GROUP BY quarter ORDER BY quarter;

-- [A3] 지역×분기 집계 원본
SELECT region, quarter, SUM(amount) AS total_amount, COUNT(*) AS fact_rows FROM sales_long GROUP BY region, quarter ORDER BY region, quarter;

-- [A4] 완성된 Wide 자료 3행
SELECT region, q1, q2, q3 FROM sales_wide ORDER BY region;

-- [A5] 열별 자료가 있는 지역 수
SELECT COUNT(*) AS regions, COUNT(q1) AS q1_nonnull, COUNT(q2) AS q2_nonnull, COUNT(q3) AS q3_nonnull FROM sales_wide;

-- [B1] PIVOT의 SUM에 대응하는 조건부 집계
SELECT region, SUM(CASE WHEN quarter='Q1' THEN amount END) AS q1, SUM(CASE WHEN quarter='Q2' THEN amount END) AS q2, SUM(CASE WHEN quarter='Q3' THEN amount END) AS q3 FROM sales_long GROUP BY region ORDER BY region;

-- [B2] SUM 대신 행 건수 COUNT
SELECT region, COUNT(CASE WHEN quarter='Q1' THEN 1 END) AS q1_rows, COUNT(CASE WHEN quarter='Q2' THEN 1 END) AS q2_rows, COUNT(CASE WHEN quarter='Q3' THEN 1 END) AS q3_rows FROM sales_long GROUP BY region ORDER BY region;

-- [B3] PIVOT AVG에 대응하는 조건부 평균
SELECT region, AVG(CASE WHEN quarter='Q1' THEN amount END) AS q1_avg, AVG(CASE WHEN quarter='Q2' THEN amount END) AS q2_avg, AVG(CASE WHEN quarter='Q3' THEN amount END) AS q3_avg FROM sales_long GROUP BY region ORDER BY region;

-- [B4] PIVOT MAX와 SUM 비교
SELECT region, MAX(CASE WHEN quarter='Q1' THEN amount END) AS q1_max, SUM(CASE WHEN quarter='Q1' THEN amount END) AS q1_sum FROM sales_long GROUP BY region ORDER BY region;

-- [B5] PIVOT IN 목록에 Q4까지 포함
SELECT region, SUM(CASE WHEN quarter='Q1' THEN amount END) AS q1, SUM(CASE WHEN quarter='Q2' THEN amount END) AS q2, SUM(CASE WHEN quarter='Q3' THEN amount END) AS q3, SUM(CASE WHEN quarter='Q4' THEN amount END) AS q4 FROM sales_long GROUP BY region ORDER BY region;

-- [C1] 추가 채널 컬럼 때문에 6개 그룹
SELECT region, channel, SUM(CASE WHEN quarter='Q1' THEN amount END) AS q1, SUM(CASE WHEN quarter='Q2' THEN amount END) AS q2, SUM(CASE WHEN quarter='Q3' THEN amount END) AS q3 FROM sales_long GROUP BY region, channel ORDER BY region, channel;

-- [C2] 지역만 남긴 명시적 투영은 3개 그룹
SELECT region, SUM(CASE WHEN quarter='Q1' THEN amount END) AS q1, SUM(CASE WHEN quarter='Q2' THEN amount END) AS q2, SUM(CASE WHEN quarter='Q3' THEN amount END) AS q3 FROM (SELECT region, quarter, amount FROM sales_long) s GROUP BY region ORDER BY region;

-- [C3] 지역별 채널 수
SELECT region, COUNT(DISTINCT channel) AS channel_groups FROM sales_long GROUP BY region ORDER BY region;

-- [C4] 서울만 채널별 PIVOT
SELECT region, channel, SUM(CASE WHEN quarter='Q1' THEN amount END) AS q1, SUM(CASE WHEN quarter='Q2' THEN amount END) AS q2 FROM sales_long WHERE region='서울' GROUP BY region, channel ORDER BY channel;

-- [C5] 채널별 소계를 다시 합치면 지역별 결과
WITH by_channel AS (SELECT region,channel,SUM(CASE WHEN quarter='Q1' THEN amount END) AS q1, SUM(CASE WHEN quarter='Q2' THEN amount END) AS q2 FROM sales_long GROUP BY region, channel) SELECT region,SUM(q1) AS q1,SUM(q2) AS q2 FROM by_channel GROUP BY region ORDER BY region;

-- [D1] UNPIVOT 입력 Wide의 9개 칸
SELECT region, q1, q2, q3 FROM sales_wide ORDER BY region;

-- [D2] UNPIVOT EXCLUDE NULLS 대응 7행
SELECT region,quarter,amount FROM (SELECT region,'Q1' AS quarter,q1 AS amount FROM sales_wide

UNION ALL SELECT region,'Q3',q3 FROM sales_wide) x WHERE amount IS NOT NULL ORDER BY region,quarter;

SELECT region,quarter,amount FROM (SELECT region,'Q1' AS quarter,q1 AS amount FROM sales_wide

UNION ALL SELECT region,'Q3',q3 FROM sales_wide) x ORDER BY region,quarter;

-- [E2] Q3: 실제 NULL 값의 존재
SELECT region,COUNT(CASE WHEN quarter='Q3' THEN 1 END) AS q3_source_rows,COUNT(CASE WHEN quarter='Q3' THEN amount END) AS q3_nonnull_values,SUM(CASE WHEN quarter='Q3' THEN amount END) AS q3_sum FROM sales_long GROUP BY region ORDER BY region;

-- [E3] ELSE 0을 쓰면 빈 칸이 0으로 바뀜
